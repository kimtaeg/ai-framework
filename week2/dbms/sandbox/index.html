    sql("INSERT INTO users (name, age) VALUES ('정글', 25)");\nsql('SELECT * FROM users WHERE id = 1');\n// → 표 + 실행 계획 (B+ 트리 인덱스 조회)\nconst [r] = db.execute('SELECT name FROM users');\nprint(r.rows);
    코드 안에서 쓸 수 있는 것
    • sql('SELECT * FROM users WHERE id = 1') — 실행 + 결과 표 출력, 결과 배열 반환
    • db.execute(sql) — 출력 없이 결과만 · 테이블 users(id, name, age, email), products(id, name, price, category), posts(id, title, author)
    • tokenize(sql) · parse(sql) — 처리기의 1·2단계
    • new BPlusTree(차수) · seedRows(db, 'users', n) · redis
    • print(…) · await sleep(ms) · time('이름', () => { … })
